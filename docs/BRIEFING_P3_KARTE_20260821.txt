5. GANZ AM ENDE der Datei, NACH dem bestehenden `<script is:inline>...</script>`, ein NEUES gebündeltes Script (NICHT is:inline) mit exakt dieser Logik:
```astro
<script>
  // Karten-Pick: Leaflet dynamisch, OSM-Tiles nur mit Klaro-Consent ("all")
  const mapPickBtn = document.getElementById("map-pick-btn");
  const mapContainer = document.getElementById("map-pick-container");
  const mapConsentHint = document.getElementById("map-consent-hint");
  const latInput = document.getElementById("lat");
  const lngInput = document.getElementById("lng");

  function hasMapConsent() {
    try { return localStorage.getItem("klaro-consent") === "all"; } catch (e) { return false; }
  }

  if (mapPickBtn && mapContainer && latInput && lngInput) {
    mapPickBtn.addEventListener("click", async function () {
      if (!hasMapConsent()) {
        if (mapConsentHint) mapConsentHint.classList.remove("hidden");
        return;
      }
      if (mapConsentHint) mapConsentHint.classList.add("hidden");
      mapContainer.classList.remove("hidden");
      const L = await import("leaflet");
      const pinIcon = L.divIcon({
        className: "",
        html: '<svg xmlns="http://www.w3.org/2000/svg" width="30" height="42" viewBox="0 0 24 24" fill="none" stroke="#0b6e4f" stroke-width="2.2" stroke-linecap="round" stroke-linejoin="round"><path d="M21 10c0 7-9 13-9 13s-9-6-9-13a9 9 0 0 1 18 0z"/><circle cx="12" cy="10" r="3" fill="#0b6e4f" stroke="none"/></svg>',
        iconSize: [30, 42],
        iconAnchor: [15, 42],
      });
      let marker = null;
      if (!window.__mapPick) {
        window.__mapPick = L.map(mapContainer, { zoomControl: true, attributionControl: true });
        L.tileLayer("https://tile.openstreetmap.org/{z}/{x}/{y}.png", {
          maxZoom: 19,
          attribution: '&copy; <a href="https://www.openstreetmap.org/copyright">OpenStreetMap</a>',
        }).addTo(window.__mapPick);
        window.__mapPick.on("click", function (e) {
          const lat = e.latlng.lat.toFixed(6);
          const lng = e.latlng.lng.toFixed(6);
          latInput.value = lat;
          lngInput.value = lng;
          if (marker) window.__mapPick.removeLayer(marker);
          marker = L.marker([lat, lng], { icon: pinIcon }).addTo(window.__mapPick);
        });
        const curLat = parseFloat(latInput.value);
        const curLng = parseFloat(lngInput.value);
        if (!isNaN(curLat) && !isNaN(curLng)) {
          window.__mapPick.setView([curLat, curLng], 13);
        } else {
          window.__mapPick.setView([47.5, 14.3], 7);
        }
      } else {
        window.__mapPick.invalidateSize();
      }
    });
  }
</script>
```
